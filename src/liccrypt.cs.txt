using System;
using System.Globalization;
using System.IO;
using System.Security.Cryptography;
using System.Text;

// standalone clone of ZooCommon.ZooLicenseHelper crypto
class P
{
    static SymmetricAlgorithm GetAlgorithm(string serial)
    {
        byte[] salt = Encoding.Unicode.GetBytes(serial.Length.ToString(CultureInfo.InvariantCulture));
        var pdb = new PasswordDeriveBytes(serial, salt);
        var aes = Aes.Create();
        aes.Key = pdb.GetBytes(32);
        aes.IV = pdb.GetBytes(16);
        return aes;
    }
    static byte[] Dec(string s, byte[] d)
    {
        var alg = GetAlgorithm(s);
        using var ms = new MemoryStream();
        using var cs = new CryptoStream(ms, alg.CreateDecryptor(), CryptoStreamMode.Write);
        cs.Write(d, 0, d.Length); cs.FlushFinalBlock(); return ms.ToArray();
    }
    static byte[] Enc(string s, byte[] d)
    {
        var alg = GetAlgorithm(s);
        using var ms = new MemoryStream();
        using var cs = new CryptoStream(ms, alg.CreateEncryptor(), CryptoStreamMode.Write);
        cs.Write(d, 0, d.Length); cs.FlushFinalBlock(); cs.Close(); return ms.ToArray();
    }
    static void Main(string[] a)
    {
        // mode: reenc <fromSerial> <toSerial> <src> <dst>
        if (a[0] == "reenc")
        {
            byte[] pt = Dec(a[1], File.ReadAllBytes(a[3]));
            File.WriteAllBytes(a[4], Enc(a[2], pt));
            Console.WriteLine("OK ptlen=" + pt.Length + " magic=" + (pt[0] == '<'));
        }
        else if (a[0] == "dec")
        {
            Console.Write(Encoding.UTF8.GetString(Dec(a[1], File.ReadAllBytes(a[2]))));
        }
    }
}
